cmake_minimum_required(VERSION 3.16)
project(NoteMDX VERSION 0.8 LANGUAGES CXX)
add_library(notemdx_core STATIC src/compiler.cpp)
target_include_directories(notemdx_core PUBLIC include)
target_compile_features(notemdx_core PUBLIC cxx_std_17)
add_executable(notemdx src/main.cpp)
target_link_libraries(notemdx PRIVATE notemdx_core)
if(MSVC)
  target_compile_options(notemdx_core PRIVATE /W4 /utf-8)
  target_compile_options(notemdx PRIVATE /W4 /utf-8)
else()
  target_compile_options(notemdx_core PRIVATE -Wall -Wextra -Wpedantic)
  find_package(Iconv REQUIRED)
  target_link_libraries(notemdx PRIVATE Iconv::Iconv)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(notemdx_tests tests/test_compiler.cpp)
  target_link_libraries(notemdx_tests PRIVATE notemdx_core)
  add_test(NAME compiler COMMAND notemdx_tests)
  find_package(Python3 COMPONENTS Interpreter QUIET)
  if(Python3_Interpreter_FOUND)
    add_test(NAME reference COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_reference.py $<TARGET_FILE:notemdx>)
    add_test(NAME il_conversion COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_il.py $<TARGET_FILE:notemdx>)
    add_test(NAME statistics COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_statistics.py $<TARGET_FILE:notemdx>)
    add_test(NAME manuals COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_manuals.py)
    add_test(NAME legacy_cli COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_legacy_cli.py $<TARGET_FILE:notemdx>)
    add_test(NAME large64k_probes COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_large64k_probes.py $<TARGET_FILE:notemdx>)
    add_test(NAME cli COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_cli.py $<TARGET_FILE:notemdx>)
  endif()
endif()
